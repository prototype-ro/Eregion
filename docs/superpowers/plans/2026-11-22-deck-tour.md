# Deck Page Tour Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A short tour of the deck page that points at the real controls, offered once and restartable, so a new player learns how Eregion works.

**Architecture:** A `tour_done` flag on the account (beside `variants_enabled`) says whether the offer still shows. Steps are plain data in `lib/tour.ts`, each naming a `data-tour` attribute on the page; steps whose target is not on screen are skipped. One client component draws a spotlight and a card beside the target, with Back, Next and Skip. The tour only points: it never clicks, edits or saves.

**Tech Stack:** Spring Boot 4 + JdbcClient + Flyway (backend), Next.js 16 + React 19 + Tailwind v4 (frontend), node:test for frontend unit tests, JUnit 5 + Testcontainers (`PostgresTest`) for backend tests.

```ts
export interface Preferences {
  variantsEnabled: boolean;
  /** The player finished or skipped the deck page tour, so it is no longer offered. */
  tourDone: boolean;
}

const DEFAULT: Preferences = { variantsEnabled: false, tourDone: false };
let cached: Promise<Preferences> | null = null;
let known: Preferences | null = null;
const listeners = new Set<(value: Preferences) => void>();

/** Asked once per page load and shared. loaded is false until the account answered. */
export function usePreferences(): Preferences & { loaded: boolean } {
  const [preferences, setPreferences] = useState<Preferences | null>(known);
  useEffect(() => {
    cached ??= api<Preferences>("/api/account/preferences")
      .then((value) => (known = value))
      .catch(() => {
        cached = null;
        return DEFAULT;
      });
    let live = true;
    void cached.then((value) => live && setPreferences(value));
    const listen = (value: Preferences) => live && setPreferences(value);
    listeners.add(listen);
    return () => {
      live = false;
      listeners.delete(listen);
    };
  }, []);
  return { ...(preferences ?? DEFAULT), loaded: preferences !== null };
}

/** Saves the settings given, then tells every component using the preferences. */
export async function savePreferences(change: Partial<Preferences>): Promise<Preferences> {
  const saved = await api<Preferences>("/api/account/preferences", { method: "PUT", body: change });
  known = saved;
  cached = Promise.resolve(saved);
  listeners.forEach((listen) => listen(saved));
  return saved;
}

export const setVariants = (enabled: boolean) => savePreferences({ variantsEnabled: enabled });
```

```tsx
"use client";

import { useCallback, useEffect, useLayoutEffect, useRef, useState } from "react";
import { createPortal } from "react-dom";
import { stepsToShow, type TourStep } from "@/lib/tour";
import { btn } from "@/lib/ui";

interface Props {
  steps: TourStep[];
  /** Called once, when the tour is finished or skipped. */
  onClose: () => void;
}

const find = (target: string) => document.querySelector<HTMLElement>(`[data-tour="${target}"]`);
const visible = (target: string) => {
  const el = find(target);
  if (!el) return false;
  const box = el.getBoundingClientRect();
  return box.width > 0 && box.height > 0;
};
const PAD = 8;
const CARD = 340;

/**
 * Points at one part of the page at a time: the page dims around it and a card says what it is for. It only points; Back,
 * Next and Skip move it, Escape closes it, and nothing on the page is clicked or changed.
 */
export function Tour({ steps, onClose }: Props) {
  const [shown] = useState(() => stepsToShow(steps, visible));
  const [at, setAt] = useState(0);
  const [box, setBox] = useState<DOMRect | null>(null);
  const next = useRef<HTMLButtonElement>(null);
  const step = shown[at];

  const measure = useCallback(() => {
    const el = step ? find(step.target) : null;
    setBox(el ? el.getBoundingClientRect() : null);
  }, [step]);

  useLayoutEffect(() => {
    const el = step ? find(step.target) : null;
    if (!el) return;
    const reduce = matchMedia("(prefers-reduced-motion: reduce)").matches;
    el.scrollIntoView({ block: "center", behavior: reduce ? "auto" : "smooth" });
    measure();
    next.current?.focus();
  }, [step, measure]);

  useEffect(() => {
    let frame = 0;
    const later = () => {
      cancelAnimationFrame(frame);
      frame = requestAnimationFrame(measure);
    };
    const key = (event: KeyboardEvent) => event.key === "Escape" && onClose();
    window.addEventListener("scroll", later, true);
    window.addEventListener("resize", later);
    window.addEventListener("keydown", key);
    return () => {
      cancelAnimationFrame(frame);
      window.removeEventListener("scroll", later, true);
      window.removeEventListener("resize", later);
      window.removeEventListener("keydown", key);
    };
  }, [measure, onClose]);

  useEffect(() => {
    if (shown.length === 0) onClose();
  }, [shown.length, onClose]);
  if (!step || typeof document === "undefined") return null;

  const last = at === shown.length - 1;
  // The card goes under the target when there is room, otherwise above it; it stays inside the window.
  const below = !box || box.bottom + 230 < window.innerHeight;
  const left = box ? Math.min(Math.max(box.left + box.width / 2 - CARD / 2, 12), window.innerWidth - CARD - 12) : 12;
  const top = box ? (below ? box.bottom + PAD + 12 : Math.max(box.top - PAD - 12 - 210, 12)) : 12;

  return createPortal(
    <div className="fixed inset-0 z-[70]">
      {/* Clicking the dimmed page does nothing, so a stray click never edits the deck mid-tour. */}
      <div className="absolute inset-0" aria-hidden="true" />
      {box && (
        <div
          aria-hidden="true"
          className="pointer-events-none absolute rounded-[20px] transition-all duration-300 ease-out motion-reduce:transition-none"
          style={{ left: box.left - PAD, top: box.top - PAD, width: box.width + PAD * 2, height: box.height + PAD * 2, boxShadow: "0 0 0 9999px rgba(0,0,0,0.55)", outline: "2px solid var(--accent-text)" }}
        />
      )}
      <div
        role="dialog"
        aria-modal="true"
        aria-labelledby="tour-title"
        aria-describedby="tour-body"
        className="absolute rounded-3xl border border-line bg-bg p-5 shadow-2xl"
        style={{ left, top, width: CARD }}
      >
        <p className="font-mono text-xs text-muted">
          {at + 1} of {shown.length}
        </p>
        <h2 id="tour-title" className="mt-1 text-xl font-medium">
          {step.title}
        </h2>
        <p id="tour-body" className="mt-2 text-[15px] leading-normal text-muted">
          {step.body}
        </p>
        <div className="mt-4 flex items-center gap-2">
          <button ref={next} type="button" onClick={() => (last ? onClose() : setAt(at + 1))} className={btn.primary}>
            {last ? "Done" : "Next"}
          </button>
          {at > 0 && (
            <button type="button" onClick={() => setAt(at - 1)} className={btn.secondary}>
              Back
            </button>
          )}
          {!last && (
            <button type="button" onClick={onClose} className={`${btn.quiet} ml-auto`}>
              Skip the tour
            </button>
          )}
        </div>
      </div>
    </div>,
    document.body,
  );
}
```
